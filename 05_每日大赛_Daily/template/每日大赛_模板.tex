\begin{solution}{Section A \textperiodcentered\ 604 Foundation Solutions}
\textbf{A.1:}\enspace $\sin x = x - \frac{x^3}{6} + o(x^3)$, $\cos x = 1 - \frac{x^2}{2} + o(x^2)$。代入分子得 $\left(x - \frac{x^3}{6}\right) - x\left(1 - \frac{x^2}{2}\right) + o(x^3) = \frac{1}{3}x^3 + o(x^3)$，故极限 $L = \lim_{x\to 0}\frac{x^3/3 + o(x^3)}{x^3} = \frac{1}{3}$。
\begin{warnbox}
\alerttag{Common Error}\enspace 若直接用等价无穷小 $\cos x \sim 1 - x^2/2$，则分子得到 $x^3/2$，导致系数错误！减法相消必须严格保留高阶项。
\end{warnbox}

\vspace{1mm}
\textbf{A.2:}\enspace 特征行列式 $|\lambda I - A| = (\lambda-2)[(\lambda-2)^2 - 1] = (\lambda-2)(\lambda-1)(\lambda-3) = 0$，解得特征值为 $\lambda_1 = 1, \lambda_2 = 2, \lambda_3 = 3$。自检：$\mathrm{tr}(A)=6=1+2+3$，$|A|=2(4-1)=6=1\times 2\times 3$，完全吻合。

\vspace{1mm}
\textbf{A.3:}\enspace 积分因子 $I(x) = e^{\int -2x dx} = e^{-x^2}$。方程变形为 $\frac{d}{dx}[y e^{-x^2}] = e^{-x^2}$。结合初值 $y(0)=0$ 积分即得：$y(x) = e^{x^2}\int_0^x e^{-t^2}dt$。
\end{solution}

\vspace{3pt}

\begin{solution}{Section B \& Section C \textperiodcentered\ 811 QM Solutions}
\textbf{B.1:}\enspace \alerttag{错误}。正好相反。Heisenberg 图像中态矢冻结 $|\psi_H\rangle = |\psi(0)\rangle$，算符随时间演化；Schrödinger 图像中才是态矢演化、算符不含时。

\vspace{1mm}
\textbf{B.2:}\enspace 物理实质是全空间概率严格守恒。空间某一闭区域内发现粒子的几率减少速率等于概率以流密度 $\vec{j}$ 流出边界的通量强度。

\vspace{1mm}
\textbf{C.1:}\enspace
(1) $[\hat{x}, \hat{H}] = \frac{1}{2m}[\hat{x}, \hat{p}^2] = \frac{i\hbar}{m}\hat{p} \implies \frac{d\hat{x}_H}{dt} = \frac{1}{i\hbar}\left(\frac{i\hbar}{m}\hat{p}_H\right) = \frac{\hat{p}_H}{m}$。\\
(2) $[\hat{p}, \hat{H}] = \frac{1}{2}m\omega^2 [\hat{p}, \hat{x}^2] = -i\hbar m\omega^2 \hat{x} \implies \frac{d\hat{p}_H}{dt} = \frac{1}{i\hbar}(-i\hbar m\omega^2 \hat{x}_H) = -m\omega^2 \hat{x}_H$。\\
(3) 对 (1) 再对时间求导：$\frac{d^2\hat{x}_H}{dt^2} = \frac{1}{m}\frac{d\hat{p}_H}{dt} = -\omega^2 \hat{x}_H \iff \frac{d^2\hat{x}_H}{dt^2} + \omega^2 \hat{x}_H = 0$。
\begin{notebox}
\alerttag{Ehrenfest's Theorem}\enspace 这与经典简谐振动牛顿第二定律形式完全一致！两边取期望值即得平均值运动规律与经典轨迹重合。
\end{notebox}
\end{solution}

\begin{solution}{Section D \textperiodcentered\ Gaussian Integral Solution}
构造极坐标二重积分：$I^2 = \iint_{\mathbb{R}^2} e^{-a(x^2+y^2)}dxdy = \int_0^{2\pi}d\theta \int_0^{+\infty} e^{-ar^2}rdr = 2\pi \left[-\frac{1}{2a}e^{-ar^2}\right]_0^\infty = \frac{\pi}{a}$。\\
因被积函数恒正，$I > 0$，故开方得解：$I = \sqrt{\frac{\pi}{a}}$。
\end{solution}
