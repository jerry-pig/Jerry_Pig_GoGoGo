\begin{problem}{Problem M1 \textbar\ 604 Calculus: auxiliary function proof}{Suggested time: 25\,min}

证明：对任意给定的实常数 $k > 0$，必存在 $\xi \in (0, 1)$，使得：
\[
  f'(\xi) + k f(\xi) = k.

要求写清辅助函数的构造思路、零点定位及罗尔定理的应用条件。

\begin{solution}{M1 \textperiodcentered\ Model Solution \& Commentary}
\textbf{Strategy:}\enspace 目标式移项为 $f'(\xi) + k[f(\xi)-1] = 0$，令 $g(x)=f(x)-1$，乘积分因子 $e^{kx}$ 构造 $F(x)=e^{kx}[f(x)-1]$。

\vspace{1mm}
\textbf{Derivation:}\enspace
令辅助函数 $F(x) = e^{kx}[f(x)-1]$。显然 $F(x) \in C[0,1]$ 且在 $(0,1)$ 内可导。\\
代入端点值：$F(0) = e^0[f(0)-1] = -1$；$F(1) = e^k[f(1)-1] = 0$。\\
由 $f(0)=0, f(1)=1$ 且 $f(x)$ 连续，根据介值定理，必存在 $c \in (0,1)$ 使得 $f(c)=1$。\\
此时在区间 $[c, 1]$ 上：$F(c) = e^{kc}[f(c)-1] = 0$，$F(1) = 0$。\\
在 $[c, 1]$ 上应用 Rolle 定理：$\exists \xi \in (c, 1) \subset (0,1)$，使得 $F'(\xi) = 0$。\\
求导：$F'(x) = k e^{kx}[f(x)-1] + e^{kx}f'(x) = e^{kx}[f'(x) + k f(x) - k]$。\\
因为 $e^{k\xi} \neq 0$，所以 $f'(\xi) + k f(\xi) = k$。证毕！

\begin{warnbox}
\alerttag{Common Error}\enspace 许多考生直接在 $[0,1]$ 用 Rolle 定理，却忽视了 $F(0)=-1 \neq F(1)=0$！必须先用介值定理找到中间根 $f(c)=1$，将区间缩小到 $[c,1]$ 才能应用罗尔定理。
\end{warnbox}
\end{solution}

\vspace{4pt}

\begin{solution}{Q1 \textperiodcentered\ Model Solution \& Commentary}
\textbf{Physical Picture:}\enspace 三维谐振子可在直角坐标中完全分离变量；奇数个阶梯算符无法将态矢原样送回，矩阵元严格按选择定则相消。

\vspace{1mm}
\textbf{Derivation:}\enspace
(1) 分离变量 $\hat{H}_0 = \hat{H}_x + \hat{H}_y + \hat{H}_z$，能谱为 $E_N = (n_x+n_y+n_z+3/2)\hbar\omega = (N+3/2)\hbar\omega$。\\
简并度为方程 $n_x+n_y+n_z=N$ 的非负整数解个数：
\[
  g_N = \sum_{n_x=0}^N (N - n_x + 1) = (N+1)(N+1) - \frac{N(N+1)}{2} = \frac{(N+1)(N+2)}{2}.
\]
(2) 坐标算符 $x_i = \sqrt{\frac{\hbar}{2m\omega}}(\hat{a}_i + \hat{a}_i^\dagger)$。微扰 $\hat{H}' \propto (\hat{a}_x+\hat{a}_x^\dagger)(\hat{a}_y+\hat{a}_y^\dagger)(\hat{a}_z+\hat{a}_z^\dagger)$。\\
基态一级修正 $E_0^{(1)} = \langle 000|\hat{H}'|000\rangle = \lambda \langle 0|x|0\rangle \langle 0|y|0\rangle \langle 0|z|0\rangle = 0$。\\
(3) 对于 $N=1$ 子空间 $\{|100\rangle, |010\rangle, |001\rangle\}$，任意矩阵元为：
\[
  H'_{ij} = \lambda \langle n_x n_y n_z | x y z | n'_x n'_y n'_z \rangle = \lambda \langle n_x|x|n'_x\rangle \langle n_y|y|n'_y\rangle \langle n_z|z|n'_z\rangle.
\]
因为跃迁需要 $\Delta n_x = \pm 1, \Delta n_y = \pm 1, \Delta n_z = \pm 1$，在 $N=1$ 子空间内（如 $|100\rangle \to |010\rangle$），只有 $x$ 和 $y$ 发生变化，而 $z$ 轴量子数均为 $0$（$\Delta n_z = 0$），导致 $\langle 0|z|0\rangle = 0$！\\
因此微扰割裂矩阵为全零矩阵 $\mathbf{H}' = \mathbf{0}_{3\times 3}$，一级微扰能量修正全部为零：$E_1^{(1)} = 0$！

\begin{notebox}
\alerttag{Checkpoint}\enspace 三维谐振子简并度 $g_0=1, g_1=3, g_2=6$。微扰项 $xyz$ 只在 $\Delta n_x, \Delta n_y, \Delta n_z$ 均为 $\pm 1$ 时非零，故一阶微扰不发生能级分裂，分裂发生在二阶微扰。
\end{notebox}
\end{solution}
